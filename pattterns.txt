⭐ 18. Hollow Pyramid
    *
   * *
  *   *
 *     *
*********
Logic
SPACES → STAR → INNER SPACE → STAR
for (let i = 0; i < n; i++) {

    let row = "";

    // spaces
    for (let j = 0; j < n - i - 1; j++) {
        row += " ";
    }

    // pyramid width
    let width = 2 * i + 1;

    for (let j = 0; j < width; j++) {

        if (j == 0 || j == width - 1 || i == n - 1) {
            row += "*";
        } else {
            row += " ";
        }
    }

    console.log(row);
}
Important

Last row completely stars.

for (let i = 1; i <= n; i++) {

    for (let j = 0; j < n - i; j++) {
        row += " ";
    }

    // increasing
    for (let j = 1; j <= i; j++) {
        row += j;
